Procudres sample code:

INCLUDE Irvine32.inc
.data 
num1 DWORD 10        ; First integer (10)  
num2 DWORD 20        ; Second integer (20)  
num3 DWORD 30        ; Third integer (30)  
result DWORD 0       ; Variable to store the result

.code
main PROC
  
call sum
  
mov eax, result    ; Move result into EAX for printing call WriteDec     
; Use Irvine32 library to print the result
  
call Crlf          
  
call WaitMsg      
  
exit

main ENDP

sum PROC
   
; Load the integers into registers and compute the sum
   
mov eax, num1      ; Load num1 into EAX

add eax, num2      ; Add num2 to EAX   
add eax, num3      ; Add num3 to EAX

mov result, eax    ; Store the result in the 'result' variable
ret                ; Return to the caller
sum ENDP


END main
